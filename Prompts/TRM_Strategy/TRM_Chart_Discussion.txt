import java.util.ArrayList;
import java.util.List;

public class TRMCalculator {

    // Result DTO (Java 8 compatible class)
    public static class Result {
        private final List<Double> sequence;
        private final double totalInvested;
        private final double nextInvestment;
        private final double bankrollRequired;

        public Result(List<Double> sequence, double totalInvested,
                      double nextInvestment, double bankrollRequired) {
            this.sequence = sequence;
            this.totalInvested = totalInvested;
            this.nextInvestment = nextInvestment;
            this.bankrollRequired = bankrollRequired;
        }

        public List<Double> getSequence() {
            return sequence;
        }

        public double getTotalInvested() {
            return totalInvested;
        }

        public double getNextInvestment() {
            return nextInvestment;
        }

        public double getBankrollRequired() {
            return bankrollRequired;
        }
    }

    public static Result calculate(double resetAmount, int maxLosses) {

        double bet = resetAmount;
        double total = 0;

        List<Double> sequence = new ArrayList<>();

        for (int i = 1; i <= maxLosses; i++) {
            sequence.add(bet);
            total += bet;
            bet = total * 2;
        }

        double bankroll = total + bet;

        return new Result(sequence, total, bet, bankroll);
    }

    public static void main(String[] args) {

        double resetAmount = 5;
        int maxLosses = 10;

        Result result = calculate(resetAmount, maxLosses);

        System.out.println("===== Teja Recovery Model (TRM) =====");
        System.out.println("Reset Amount          : $" + resetAmount);
        System.out.println("Maximum Losses        : " + maxLosses);

        System.out.println("\nInvestment Sequence:");

        for (int i = 0; i < result.getSequence().size(); i++) {
            System.out.printf("Step %2d : $%,.0f%n",
                    i + 1,
                    result.getSequence().get(i));
        }

        System.out.println("\n===== Summary =====");
        System.out.printf("Total Invested        : $%,.0f%n", result.getTotalInvested());
        System.out.printf("Next Investment       : $%,.0f%n", result.getNextInvestment());
        System.out.printf("Bankroll Required     : $%,.0f%n", result.getBankrollRequired());

        System.out.println("\nClosed Form Formula Check");
        double formulaBankroll = resetAmount * Math.pow(3, maxLosses);
        System.out.printf("S × 3^N               : $%,.0f%n", formulaBankroll);
    }
}

import java.math.BigDecimal;
import java.math.MathContext;
import java.math.RoundingMode;
import java.util.ArrayList;
import java.util.List;

public class TRMComparison {

    private static final MathContext MC =
            new MathContext(20, RoundingMode.HALF_UP);

    // ============================================================
    // COMMON RESULT FOR ORIGINAL TRM
    // ============================================================

    public record TRMResult(
            List<BigDecimal> investmentSequence,
            BigDecimal totalInvested,
            BigDecimal nextInvestment,
            BigDecimal bankrollRequired
    ) {
    }

    // ============================================================
    // COMMON RESULT FOR STOCK-BASED TRM V1 / V2
    // ============================================================

    public record StockStep(
            int round,
            BigDecimal stockPrice,
            BigDecimal investment,
            BigDecimal sharesBought,
            BigDecimal totalInvested,
            BigDecimal totalShares,
            BigDecimal averageCost,
            BigDecimal marketValue,
            BigDecimal unrealizedPnL
    ) {
    }

    public record StockTRMResult(
            List<StockStep> steps,
            BigDecimal totalInvested,
            BigDecimal totalShares,
            BigDecimal averageCost,
            BigDecimal currentPrice,
            BigDecimal marketValue,
            BigDecimal unrealizedPnL,
            BigDecimal nextInvestment
    ) {
    }

    // ============================================================
    // TRM - ORIGINAL MODEL
    //
    // Rule:
    // First investment = S
    // Next investment = 2 × cumulative invested amount
    //
    // Example:
    // 5 -> 10 -> 30 -> 90 -> 270 -> 810 ...
    //
    // Total after N losses:
    // T(N) = S × 3^(N-1)
    //
    // Bankroll required to survive N losses and make next bet:
    // S × 3^N
    // ============================================================

    static class TRM {

        public static TRMResult calculate(
                BigDecimal resetAmount,
                int maxLosses
        ) {

            validatePositive(resetAmount, "resetAmount");

            if (maxLosses < 1) {
                throw new IllegalArgumentException(
                        "maxLosses must be >= 1"
                );
            }

            BigDecimal bet = resetAmount;
            BigDecimal totalInvested = BigDecimal.ZERO;

            List<BigDecimal> sequence = new ArrayList<>();

            for (int i = 1; i <= maxLosses; i++) {

                // Current investment
                sequence.add(bet);

                // Add current investment to cumulative amount
                totalInvested = totalInvested
                        .add(bet, MC);

                // Next investment = 2 × cumulative investment
                bet = totalInvested
                        .multiply(BigDecimal.valueOf(2), MC);
            }

            // Amount required to place the next investment
            BigDecimal bankrollRequired =
                    totalInvested.add(bet, MC);

            return new TRMResult(
                    sequence,
                    totalInvested,
                    bet,
                    bankrollRequired
            );
        }
    }

                );

                    "initialPrice"

                );
            }

            List<StockStep> steps = new ArrayList<>();

            BigDecimal currentPrice = initialPrice;

            BigDecimal totalInvested =
                    initialInvestment;

            BigDecimal totalShares =
                    initialInvestment.divide(

                            MC
                    );

            // ----------------------------------------------------
            // INITIAL PURCHASE
            // ----------------------------------------------------

            BigDecimal averageCost =
                    calculateAverageCost(
                            totalInvested,
                            totalShares
                    );

            BigDecimal marketValue =
                    totalShares.multiply(
                            currentPrice,
                            MC
                    );

            BigDecimal pnl =
                    marketValue.subtract(

                            MC
                    );

            steps.add(
                    new StockStep(
                            1,

            // ----------------------------------------------------
            // LOSS / RECOVERY ROUNDS
            // ----------------------------------------------------

            for (int round = 2;

                BigDecimal remainingPercent =
                        BigDecimal.ONE.subtract(

                        currentPrice.multiply(

# TRM V3 — Profit Retention + Loss Recovery Reversal

TRM V3 has **two separate paths**:

| Path       | Action                                                                                         |
| ---------- | ---------------------------------------------------------------------------------------------- |
| **Profit** | Keep a small portion of the realized profit as a live position and sell the rest               |
| **Loss**   | Calculate the actual loss and add a configurable recovery amount to the next reversal/re-entry |

The important correction is that **profit/loss percentage, rupee amount, and share quantity must never be mixed dimensionally**.

---

## 1. TRM V3 — Profit Scenario

Assume:

* Buy price = **₹10**
* Quantity = **100 shares**
* Initial investment = **₹1,000**
* Profit = **15%**
* Retention factor = **15%**

### Step 1 — Initial investment

$$
10\times100=₹1,000
$$

### Step 2 — Stock rises 15%

$$
10\times(1+0.15)=₹11.50
$$

### Step 3 — Profit

$$
₹1,000\times15\%=₹150
$$

So:

$$
\boxed{\text{Profit}=₹150}
$$

### Step 4 — Retain only 15% of that profit

$$
₹150\times15\%=₹22.50
$$

So the **live-position value** is:

$$
\boxed{₹22.50}
$$

### Step 5 — Convert retained value into shares

This is where your earlier calculation needed correction.

You must use the **current stock price ₹11.50**, not the original ₹10:

$$
\text{Retained Shares}
=
\frac{₹22.50}{₹11.50}
$$

$$
=1.9565
$$

So:

$$
\boxed{\text{Retain approximately 1.96 shares}}
$$

The remaining shares:

$$
100-1.9565=98.0435
$$

So:

$$
\boxed{\text{Sell approximately 98.04 shares}}
$$

| ------------------- | --------: |
| Entry price         |    ₹10.00 |
| Quantity            |       100 |
| Initial investment  |    ₹1,000 |
| Profit %            |       15% |

# 3. Next Reversal Investment

This is an important part of V3.

You need two values:

* `baseReversalInvestment`
* `recoveryAmount`

Then:

$$
\boxed{
NextReversalInvestment
=
BaseReversalInvestment
+
RecoveryAmount
}
$$

For example, suppose your normal reversal investment is ₹100:

$$
₹100+₹22.50=₹122.50
$$

Therefore:

$$
\boxed{\text{Next Reversal Investment}=₹122.50}
$$

This is the cleanest interpretation of the rule you described.

# 4. Complete TRM V3 Formula

## Profit side

### Profit amount

$$
P=I\times p
$$

where:

* \(I\) = initial investment
* \(p\) = profit percentage

### Retained profit value

$$
R=P\times r
$$

where \(r\) is the retention factor.

### Retained quantity

$$
\boxed{
Q_{retain}=\frac{R}{P_{current}}
}
$$

### Quantity to sell

$$
\boxed{
Q_{sell}=Q_{original}-Q_{retain}
}
$$

---

## Loss side

### Loss amount

$$
L=I\times l
$$

where \(l\) is the loss percentage.

### Recovery addition

$$
R_{loss}=L\times f
$$

where \(f\) is the configurable recovery factor.

### Next reversal amount

$$
\boxed{
A_{next}=A_{base}+R_{loss}
}
$$

---

import java.math.BigDecimal;
import java.math.RoundingMode;

public class TRMV3 {

    private static final int SCALE = 8;
    private static final RoundingMode ROUNDING = RoundingMode.HALF_UP;

    public record ProfitResult(
            BigDecimal entryPrice,
            BigDecimal originalQuantity,
            BigDecimal initialInvestment,
            BigDecimal profitPercent,
            BigDecimal currentPrice,
            BigDecimal profitAmount,
            BigDecimal retentionFactor,
            BigDecimal retainedProfitValue,
            BigDecimal retainedQuantity,
            BigDecimal quantityToSell
    ) {}

    public record LossResult(
            BigDecimal entryPrice,
            BigDecimal originalQuantity,
            BigDecimal initialInvestment,
            BigDecimal lossPercent,
            BigDecimal currentPrice,
            BigDecimal lossAmount,
            BigDecimal recoveryFactor,
            BigDecimal recoveryAmount,
            BigDecimal baseReversalInvestment,
            BigDecimal nextReversalInvestment
    ) {}

    // ============================================================
    // PROFIT SCENARIO
    // ============================================================

    public static ProfitResult calculateProfit(
            BigDecimal entryPrice,
            BigDecimal quantity,
            BigDecimal profitPercent,
            BigDecimal retentionFactor
    ) {

        validatePositive(entryPrice, "entryPrice");
        validatePositive(quantity, "quantity");
        validatePercentage(profitPercent, "profitPercent");
        validatePercentage(retentionFactor, "retentionFactor");

        // Initial investment
        BigDecimal initialInvestment =
                entryPrice.multiply(quantity);

        // Current stock price after profit
        BigDecimal currentPrice =
                entryPrice.multiply(
                        BigDecimal.ONE.add(profitPercent)
                );

        // Actual profit in currency
        BigDecimal profitAmount =
                initialInvestment.multiply(profitPercent);

        // Amount of profit to keep alive
        BigDecimal retainedProfitValue =
                profitAmount.multiply(retentionFactor);

        // IMPORTANT:
        // Use CURRENT PRICE to convert retained value into shares.
        BigDecimal retainedQuantity =
                retainedProfitValue.divide(
                        currentPrice,
                        SCALE,
                        ROUNDING
                );

        // Remaining shares are sold
        BigDecimal quantityToSell =
                quantity.subtract(retainedQuantity);

        return new ProfitResult(
                entryPrice,
                quantity,
                initialInvestment,
                profitPercent,
                currentPrice,
                profitAmount,
                retentionFactor,
                retainedProfitValue,
                retainedQuantity,
                quantityToSell
        );
    }

    // ============================================================
    // LOSS SCENARIO
    // ============================================================

    public static LossResult calculateLoss(
            BigDecimal entryPrice,
            BigDecimal quantity,
            BigDecimal lossPercent,
            BigDecimal recoveryFactor,
            BigDecimal baseReversalInvestment
    ) {

        validatePositive(entryPrice, "entryPrice");
        validatePositive(quantity, "quantity");
        validatePercentage(lossPercent, "lossPercent");
        validatePercentage(recoveryFactor, "recoveryFactor");
        validatePositive(
                baseReversalInvestment,
                "baseReversalInvestment"
        );

        // Initial investment
        BigDecimal initialInvestment =
                entryPrice.multiply(quantity);

        // Current price after loss
        BigDecimal currentPrice =
                entryPrice.multiply(
                        BigDecimal.ONE.subtract(lossPercent)
                );

        // Actual loss
        BigDecimal lossAmount =
                initialInvestment.multiply(lossPercent);

        // Recovery amount
        BigDecimal recoveryAmount =
                lossAmount.multiply(recoveryFactor);

        // Next reversal investment
        BigDecimal nextReversalInvestment =
                baseReversalInvestment.add(recoveryAmount);

        return new LossResult(
                entryPrice,
                quantity,
                initialInvestment,
                lossPercent,
                currentPrice,
                lossAmount,
                recoveryFactor,
                recoveryAmount,
                baseReversalInvestment,
                nextReversalInvestment
        );
    }

    // ============================================================
    // MAIN
    // ============================================================

That gives us a clean, dimensionally correct definition of **TRM V3** that can sit alongside TRM, TRM V1, and TRM V2 in your project.

One critical correction now locked into V3

For your example:

₹10 × 100 = ₹1,000

At 15% profit:

Profit = ₹150

At 15% retention:

₹150 × 15% = ₹22.50

Then:

₹22.50 ÷ ₹11.50 = 1.9565 shares retained

So the correct chain is:

₹1,000 → ₹150 profit → ₹22.50 retained value → 1.96 shares retained.

That is the version I would now treat as the official TRM V3 specification for your project.
